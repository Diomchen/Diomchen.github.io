63        sampling_params: SamplingParams | list[SamplingParams],
64        use_tqdm: bool = True,
65    ) -> list[str]:
66        pbar = tqdm(total=len(prompts), desc="Generating", dynamic_ncols=True, disable=not use_tqdm)
67        if not isinstance(sampling_params, list):
68            sampling_params = [sampling_params] * len(prompts)
69        for prompt, sp in zip(prompts, sampling_params):
70            self.add_request(prompt, sp)
71        outputs = {}
72        prefill_throughput = decode_throughput = 0.
73        while not self.is_finished():
74            t = perf_counter()
75            output, num_tokens = self.step()
76            if num_tokens > 0:
77                prefill_throughput = num_tokens / (perf_counter() - t)
78            else:
79                decode_throughput = -num_tokens / (perf_counter() - t)
80            pbar.set_postfix({
81                "Prefill": f"{int(prefill_throughput)}tok/s",
82                "Decode": f"{int(decode_throughput)}tok/s",
83            })
84            for seq_id, token_ids in output:
85                outputs[seq_id] = token_ids
86                pbar.update(1)
87        pbar.close()
88        outputs = [outputs[seq_id] for seq_id in sorted(outputs.keys())]
89        outputs = [{"text": self.tokenizer.decode(token_ids), "token_ids": token_ids} for token_ids in outputs]
