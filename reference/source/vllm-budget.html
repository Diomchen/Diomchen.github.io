809                # This is (num_computed_tokens + 1) - (num_output_placeholders - 1).
810                # Since output placeholders are also included in the computed tokens
811                # count, we subtract (num_output_placeholders - 1) to remove any draft
812                # tokens, so that we can be sure no further steps are needed even if
813                # they are all rejected.
814                and request.num_computed_tokens + 2 - request.num_output_placeholders

816            ):
817                # Async scheduling: Avoid scheduling an extra step when we are sure that
818                # the previous step has reached request.max_tokens. We don't schedule
819                # partial draft tokens since this prevents uniform decode optimizations.
