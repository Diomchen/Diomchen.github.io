1969        # where M is the max_model_len.

1971            positions_np + req_indices * self.input_batch.token_ids_cpu.shape[1]

1975        # NOTE(woosuk): We use torch.index_select instead of np.take here
1976        # because torch.index_select is much faster than np.take for large
1977        # tensors.
1978        torch.index_select(
1979            self.input_batch.token_ids_cpu_tensor.flatten(),
1980            0,
1981            token_indices_tensor,
1982            out=self.input_ids.cpu[:total_num_scheduled_tokens],

2017                    continue

2021                actual_end = min(end_pos, req_embeds.shape[0])
2022                actual_num_sched = actual_end - start_pos

2024                if actual_num_sched > 0:

2039
2040        # Compute optimistic seq_lens (assumes all draft tokens from previous
2041        # iteration accepted). Store in optimistic_seq_lens_cpu for use by
2042        # _build_attention_metadata (max_seq_len) and discard_request_mask.
2043        # seq_lens (GPU) will be computed later using the same optimistic values.
2044        torch.add(

2047            out=self.optimistic_seq_lens_cpu[:num_reqs],
2048        )
2049        self.optimistic_seq_lens_cpu[num_reqs:].fill_(0)

2051        # Build prev_positions mapping: current pos -> prev pos (-1 if new).

2053        prev_req_id_to_index = self.input_batch.prev_req_id_to_index
2054        self._compute_prev_positions(num_reqs)

2056        num_tokens = [self.requests[r].num_tokens for r in self.input_batch.req_ids]
2057        num_tokens_np = np.array(num_tokens, dtype=np.int32)

2059        # Record which requests should not be sampled,
2060        # so that we could clear the sampled tokens before returning
2061        self.discard_request_mask.np[:num_reqs] = (

2063        )
