79        q = q.view(-1, self.num_heads, self.head_dim)
80        k = k.view(-1, self.num_kv_heads, self.head_dim)
81        v = v.view(-1, self.num_kv_heads, self.head_dim)
82        if not self.qkv_bias:

84            k = self.k_norm(k)
85        q, k = self.rotary_emb(positions, q, k)
86        o = self.attn(q, k, v)
87        output = self.o_proj(o.flatten(1, -1))
88        return output
